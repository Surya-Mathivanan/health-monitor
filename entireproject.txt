LIFECARE WELLNESS — COMPLETE REACT NATIVE REBUILD PROMPT
=========================================================

Build a full React Native mobile application called "LifeCare Wellness" — a health monitoring and client management system for wellness coaches and advisors. Use Expo (with Expo Router for navigation), Supabase as the backend (auth + database + realtime), React Query (@tanstack/react-query) for data fetching, Zod + React Hook Form for form validation, and NativeWind (TailwindCSS for React Native) for styling.

--- TABLE: users (mirrors auth.users) ---
CREATE TABLE public.users (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  email TEXT UNIQUE NOT NULL,
  display_name TEXT NOT NULL DEFAULT '',
  avatar_url TEXT,
  role TEXT NOT NULL DEFAULT 'staff' CHECK (role IN ('admin', 'staff', 'co_partner')),
  position TEXT, personal_number TEXT, degree TEXT, experience TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

--- TABLE: wellness_programs ---
CREATE TABLE public.wellness_programs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  description TEXT,
  target_metrics JSONB DEFAULT '{}',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

--- TABLE: clients ---
CREATE TABLE public.clients (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  full_name TEXT NOT NULL,
  mobile TEXT NOT NULL,
  age INTEGER NOT NULL CHECK (age BETWEEN 1 AND 120),
  gender TEXT NOT NULL CHECK (gender IN ('male','female','other')),
  height_cm NUMERIC(5,1) NOT NULL,
  weight_kg NUMERIC(5,1) NOT NULL,
  assigned_program_id UUID REFERENCES public.wellness_programs(id) ON DELETE SET NULL,
  created_by UUID NOT NULL REFERENCES public.users(id) ON DELETE RESTRICT,
  status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active','inactive')),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()

--- TABLE: health_reports ---
CREATE TABLE public.health_reports (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  client_id UUID NOT NULL REFERENCES public.clients(id) ON DELETE CASCADE,
  recorded_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  body_fat_pct NUMERIC(5,2) NOT NULL,
  visceral_fat NUMERIC(4,1) NOT NULL,
  bmi NUMERIC(4,1) NOT NULL,
  resting_bmr INTEGER NOT NULL,
  body_age INTEGER NOT NULL,
  skeletal_muscle_mass NUMERIC(5,2) NOT NULL,

  visceral_flag TEXT NOT NULL DEFAULT 'normal' CHECK (visceral_flag IN ('normal','high','very_high')),
  logged_by UUID NOT NULL REFERENCES public.users(id) ON DELETE RESTRICT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

--- TABLE: call_logs ---
CREATE TABLE public.call_logs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

--- TABLE: reminders ---
CREATE TABLE public.reminders (

--- TABLE: co_partner_invitations ---
CREATE TABLE public.co_partner_invitations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  inviter_id UUID NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
  invitee_email TEXT NOT NULL,

  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

--- INDEXES ---

CREATE INDEX idx_clients_created_by ON public.clients(created_by);
CREATE INDEX idx_health_reports_client ON public.health_reports(client_id, recorded_at DESC);
CREATE INDEX idx_call_logs_client ON public.call_logs(client_id, called_at DESC);

CREATE INDEX idx_reminders_assigned ON public.reminders(assigned_to, status);

CREATE INDEX idx_invitations_email ON public.co_partner_invitations(invitee_email, status);

--- RLS POLICIES ---
Enable RLS on all tables. Key rules:

1. users: SELECT for any authenticated user. INSERT only own row (id = auth.uid()). UPDATE own row.

2. clients: SELECT where created_by = auth.uid() OR user is admin OR created_by is an accepted co-partner of the viewer (check co_partner_invitations). INSERT if authenticated. UPDATE if authenticated. DELETE only admin.

3. health_reports, call_logs, staff_notes: SELECT only for clients the current user can see (same logic as clients). INSERT own records only.

4. reminders: SELECT where assigned_to = auth.uid() OR client is supervised. INSERT if authenticated. UPDATE own or admin.

5. co_partner_invitations: Inviters see sent invitations. Invitees see received invitations (by email or invitee_id). Any authenticated user can INSERT (inviter_id = auth.uid()). Invitees can UPDATE status.

--- SECURITY DEFINER FUNCTIONS ---

-- Register new co-partner (bypasses RLS):
CREATE OR REPLACE FUNCTION public.register_new_co_partner(
  p_user_id UUID, p_email TEXT, p_display_name TEXT, p_inviter_id UUID
) RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
BEGIN
  INSERT INTO public.users (id, email, display_name, role)
  VALUES (p_user_id, p_email, p_display_name, 'co_partner')
  ON CONFLICT (id) DO UPDATE SET role='co_partner', display_name=EXCLUDED.display_name, email=EXCLUDED.email;
  INSERT INTO public.co_partner_invitations (inviter_id, invitee_email, invitee_id, status)
  VALUES (p_inviter_id, p_email, p_user_id, 'accepted')
  ON CONFLICT (inviter_id, invitee_email) DO UPDATE SET invitee_id=EXCLUDED.invitee_id, status='accepted', updated_at=NOW();
END; $$;

-- Accept co-partner invitation:
CREATE OR REPLACE FUNCTION public.accept_co_partner_invitation(p_invitation_id UUID)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE v_email TEXT;
BEGIN
  SELECT email INTO v_email FROM public.users WHERE id = auth.uid();
  UPDATE public.co_partner_invitations SET status='accepted', invitee_id=auth.uid(), updated_at=NOW()
  WHERE id=p_invitation_id AND invitee_email=v_email AND status='pending';
END; $$;

-- Decline invitation:
CREATE OR REPLACE FUNCTION public.decline_co_partner_invitation(p_invitation_id UUID)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE v_email TEXT;
BEGIN
  SELECT email INTO v_email FROM public.users WHERE id = auth.uid();
  UPDATE public.co_partner_invitations SET status='declined', updated_at=NOW()
  WHERE id=p_invitation_id AND invitee_email=v_email AND status='pending';
END; $$;

Also add a trigger on auth.users INSERT to auto-create the public.users row if not done by the client.

IMPORTANT Supabase settings:
- Disable "Confirm email" in Authentication → Providers → Email (allows instant account creation)

═══════════════════════════════════════════════════════════
DATA ISOLATION RULES (CRITICAL)
═══════════════════════════════════════════════════════════

Co-partners can ONLY see:
- Clients they personally created (created_by = their user id)
- Their own call logs, health reports, reminders, staff notes

Supervisors (users who invited co-partners) can see:
- Their own clients
- Clients of their accepted co-partners

Nobody can see another user's data unless there is an accepted co_partner_invitation relationship.

These rules are enforced at the database level via RLS policies (not just in the UI).
